cmake_minimum_required(VERSION 3.21)
project(QtShapeEditor VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Sql)
add_library(editor_core STATIC
    src/database.cpp src/database.h
    src/shapeitem.cpp src/shapeitem.h
    src/shapemodel.cpp src/shapemodel.h
    src/mainwindow.cpp src/mainwindow.h)
target_include_directories(editor_core PUBLIC src)
target_link_libraries(editor_core PUBLIC Qt6::Widgets Qt6::Sql)
if(MSVC)
    target_compile_options(editor_core PUBLIC /utf-8 /W4)
else()
    target_compile_options(editor_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
qt_add_executable(QtShapeEditor WIN32 MACOSX_BUNDLE src/main.cpp)
target_link_libraries(QtShapeEditor PRIVATE editor_core)
set_target_properties(QtShapeEditor PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Редактор фигур")
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(editor_tests tests/editor_tests.cpp)
    target_link_libraries(editor_tests PRIVATE editor_core Qt6::Test)
    add_test(NAME editor_tests COMMAND editor_tests -platform offscreen)
endif()
